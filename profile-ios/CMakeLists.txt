cmake_minimum_required(VERSION 3.20)
project(CardboardProfileTests LANGUAGES CXX)
set(FLAT_HEADSET_SOURCE "" CACHE PATH "Pinned receiver source")
if(NOT EXISTS "${FLAT_HEADSET_SOURCE}/ios/Viewer/Viewer.cpp")
  message(FATAL_ERROR "Set FLAT_HEADSET_SOURCE to the reviewed Flat Headset checkout")
endif()
add_subdirectory("${FLAT_HEADSET_SOURCE}/ios/Viewer" viewer)
add_executable(scan-test Tests/ScanTest.cpp)
target_include_directories(scan-test PRIVATE Sources "${FLAT_HEADSET_SOURCE}/ios/Viewer")
target_link_libraries(scan-test PRIVATE viewer)
target_compile_options(scan-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME scan-policy COMMAND scan-test)
